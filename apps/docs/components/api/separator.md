| icon? | IconProps['name'] | - | Display an icon on the separator. |
| avatar? | AvatarProps | - | Display an avatar on the separator. |
| color? | Separator['variants']['color'] | 'neutral' |  |
| size? | Separator['variants']['size'] | 'xs' |  |
| type? | Separator['variants']['type'] | 'solid' |  |
| orientation? | Separator['variants']['orientation'] | 'horizontal' | The orientation of the separator. |
| position? | Separator['variants']['position'] | 'center' | The position of the content. |
| class? | any | - |  |
| ui? | Separator['slots'] | - |  |
